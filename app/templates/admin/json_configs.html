Usage in Stored Procedures

After syncing, stored procedures can read configs using the helper function:

-- Get full config
SELECT fn_get_config('allocation_engine_config');

-- Get a nested value
SELECT fn_get_config('allocation_engine_config') -> 'source_tables' -> 'proc_inst_data' ->> 'account_id_column';

-- Use in a procedure
v_cfg := fn_get_config('allocation_engine_config');
v_acct_col := v_cfg -> 'source_tables' -> v_source_table ->> 'account_id_column';
